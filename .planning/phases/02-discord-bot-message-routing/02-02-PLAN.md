**Access control — cog interaction_check (AUTH-01, AUTH-04). This is the PRIMARY security gate. default_permissions is defense-in-depth UI hiding only (addresses review concern #3):**
```python
@app_commands.default_permissions(administrator=True)  # UI HIDING ONLY — not authoritative. See interaction_check below.
class GeneralCog(commands.Cog):
    """Phase 2 slash commands. Access control enforced via interaction_check (primary)
    and default_permissions (UI hiding, defense-in-depth only)."""

    def __init__(self, bot):
        self.bot = bot

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        """PRIMARY security gate for all commands in this cog.
        AUTH-01: user allowlist. AUTH-04: guild allowlist (backup to global tree check).
        NOTE: default_permissions(administrator=True) above is UI hiding only —
        it makes commands invisible to non-admin Discord users but does NOT enforce
        authorization. This method is the real enforcement."""
        # AUTH-04: guild allowlist (also checked by global tree.interaction_check from Plan 01)
        if interaction.guild_id not in self.bot.bot_config.allowed_guild_ids:
            return False  # silently ignore unauthorized guild
        # AUTH-01: user allowlist — this is the primary user-level auth gate
        if interaction.user.id not in self.bot.bot_config.allowed_user_ids:
            await interaction.response.send_message(
                "You are not authorized to use this bot.", ephemeral=True
            )
            return False
        return True
```

```python
@app_commands.command(name="status", description="Show all connected agents")
async def status(self, interaction: discord.Interaction):
    agents = self.bot.agent_manager.agents
    channel_mapping = self.bot.bot_config.channel_mapping

    embed = discord.Embed(title="Agent Status", color=0x2ECC71 if agents else 0x95A5A6)

    if not agents and not channel_mapping:
        embed.description = "No agents configured or connected"
        embed.set_footer(text="0 agent(s)")
        await interaction.response.send_message(embed=embed)  # D-10: visible
        return

    # Show connected agents
    for name, info in agents.items():
        seconds_ago = int(time.time() - info.last_pong)
        has_channel = self.bot._find_channel_for_agent(name) is not None
        status_note = "Mapped" if has_channel else "No channel mapped"
        embed.add_field(
            name=f"\U0001f7e2 {name}",
            value=(
                f"**Platform:** {info.platform}\n"
                f"**Status:** Online ({status_note})\n"
                f"**Last seen:** {seconds_ago}s ago"
            ),
            inline=False,
        )

    # Show configured-but-offline agents (in channel_mapping but not in agents dict)
    configured_machines = set(channel_mapping.values())
    connected_machines = set(agents.keys())
    offline_machines = configured_machines - connected_machines
    for machine_name in sorted(offline_machines):
        embed.add_field(
            name=f"\U0001f534 {machine_name}",
            value="**Status:** Offline (configured)",
            inline=False,
        )

    total = len(agents) + len(offline_machines)
    embed.set_footer(text=f"{len(agents)} online, {len(offline_machines)} offline — {total} agent(s) total")
    await interaction.response.send_message(embed=embed)  # D-10: visible, not ephemeral
```

```python
@app_commands.command(name="reload", description="Hot-reload config.json")
async def reload(self, interaction: discord.Interaction):
    try:
        await self.bot.bot_config.reload_config()
        # Re-sync commands to any new guilds added in config
        for guild_id in self.bot.bot_config.allowed_guild_ids:
            guild_obj = discord.Object(id=guild_id)
            self.bot.tree.copy_global_to(guild=guild_obj)
            await self.bot.tree.sync(guild=guild_obj)
        await interaction.response.send_message("Config reloaded successfully.", ephemeral=True)
        logger.info(f"Config reloaded by {interaction.user}")
    except Exception as e:
        await interaction.response.send_message(f"Reload failed: {e}", ephemeral=True)
        logger.error(f"Config reload failed: {e}")
```

```python
async def _auto_create_channel(self, machine_name: str, platform: str):
    """Auto-create a Discord channel for a newly connected agent with no mapping.
    D-14: updates config.json with new mapping via save_config (lock-protected).
    ROUT-02 note: static config is extended at runtime by auto-creation, config.json
    remains the single source of truth."""
    if not self.bot_config.allowed_guild_ids:
        logger.warning(f"No allowed guilds configured, cannot auto-create channel for {machine_name}")
        return

    # Use first allowed guild (RESEARCH Open Question 2 resolution)
    guild_id = self.bot_config.allowed_guild_ids[0]
    guild = self.get_guild(guild_id)
    if guild is None:
        logger.error(f"Guild {guild_id} not found in cache, cannot auto-create channel")
        return

    # Sanitize machine_name for Discord channel name
    channel_name = machine_name.lower().replace(" ", "-").replace("_", "-")
    channel_name = "".join(c for c in channel_name if c.isalnum() or c == "-")
    if not channel_name:
        channel_name = "unnamed-agent"

    # Duplicate prevention: check if channel with same name already exists
    existing = discord.utils.get(guild.text_channels, name=channel_name)
    if existing:
        # Map to existing channel instead of creating duplicate
        self.bot_config.channel_mapping[str(existing.id)] = machine_name
        await self.bot_config.save_config()
        try:
            await existing.send(f"\U0001f7e2 **{machine_name}** connected ({platform})\n_Mapped to existing channel._")
        except Exception as e:
            logger.error(f"Failed to send notification in existing channel for {machine_name}: {e}")
        logger.info(f"Mapped agent {machine_name} to existing channel #{channel_name} (ID: {existing.id})")
        return

    try:
        channel = await guild.create_text_channel(channel_name)
        logger.info(f"Auto-created channel #{channel_name} (ID: {channel.id}) for agent {machine_name}")

        # Update config mapping and persist
        self.bot_config.channel_mapping[str(channel.id)] = machine_name
        await self.bot_config.save_config()

        # Send connect notification in new channel
        await channel.send(f"\U0001f7e2 **{machine_name}** connected ({platform})\n_Channel auto-created for this machine._")

    except discord.Forbidden:
        logger.error(f"Missing manage_channels permission in guild {guild_id}, cannot auto-create channel for {machine_name}")
    except Exception as e:
        logger.error(f"Failed to auto-create channel for {machine_name}: {e}")
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Discord user -> Slash command | User identity from Discord, checked against allowlist via interaction_check (primary) and default_permissions (UI hiding only) |
| Bot -> Guild channel | Bot creates channels; needs manage_channels permission |
| Config.json -> Bot | Reload reads local file; lock-protected against concurrent save |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-05 | Elevation of Privilege | cog interaction_check | mitigate | interaction_check is PRIMARY auth gate verifying user.id in allowed_user_ids (AUTH-01); default_permissions is UI hiding only, NOT authoritative (documented in code comments) |
| T-02-06 | Spoofing | interaction.user.id | accept | Discord guarantees user identity via OAuth; we trust Discord's user.id |
| T-02-07 | Denial of Service | /reload command | mitigate | /reload only available to users in allowed_user_ids; uses async lock to prevent concurrent config corruption; guild re-sync bounded by guild count |
| T-02-08 | Tampering | auto-channel config write | mitigate | asyncio.Lock prevents corruption from concurrent save/reload. Duplicate prevention avoids channel spam. Config.json is local, no remote write. |
| T-02-09 | Information Disclosure | /status agent info | accept | Only visible to allowed users in allowed guilds; agent name and platform are not sensitive |
| T-02-12 | Denial of Service | auto-channel creation | mitigate | Duplicate channel name check prevents channel spam from reconnecting agents with unstable connections |
</threat_model>
