**Access control — global interaction check on the cog (D-15, AUTH-01, AUTH-04):**
```python
class GeneralCog(commands.Cog):
    def __init__(self, bot):
        self.bot = bot

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        """Global check for all commands in this cog. AUTH-01 + AUTH-04."""
        # AUTH-04: guild allowlist
        if interaction.guild_id not in self.bot.bot_config.allowed_guild_ids:
            return False  # silently ignore unauthorized guild
        # AUTH-01: user allowlist
        if interaction.user.id not in self.bot.bot_config.allowed_user_ids:
            await interaction.response.send_message(
                "You are not authorized to use this bot.", ephemeral=True
            )
            return False
        return True
```

```python
@app_commands.command(name="reload", description="Hot-reload config.json")
async def reload(self, interaction: discord.Interaction):
    try:
        self.bot.bot_config.reload_config()
        # Re-sync commands to any new guilds
        for guild_id in self.bot.bot_config.allowed_guild_ids:
            guild_obj = discord.Object(id=guild_id)
            self.bot.tree.copy_global_to(guild=guild_obj)
            await self.bot.tree.sync(guild=guild_obj)
        await interaction.response.send_message("Config reloaded successfully.", ephemeral=True)
        logger.info(f"Config reloaded by {interaction.user}")
    except Exception as e:
        await interaction.response.send_message(f"Reload failed: {e}", ephemeral=True)
        logger.error(f"Config reload failed: {e}")
```

```python
async def _auto_create_channel(self, machine_name: str, platform: str):
    """Auto-create a Discord channel for a newly connected agent with no mapping."""
    if not self.bot_config.allowed_guild_ids:
        logger.warning(f"No allowed guilds configured, cannot auto-create channel for {machine_name}")
        return

    # Use first allowed guild (Open Question 2 from RESEARCH — default to first)
    guild_id = self.bot_config.allowed_guild_ids[0]
    guild = self.get_guild(guild_id)
    if guild is None:
        logger.error(f"Guild {guild_id} not found in cache, cannot auto-create channel")
        return

    # Sanitize machine_name for Discord channel name (lowercase, hyphens only)
    channel_name = machine_name.lower().replace(" ", "-").replace("_", "-")
    # Discord channel names: 1-100 chars, lowercase, no spaces, alphanumeric + hyphens
    channel_name = "".join(c for c in channel_name if c.isalnum() or c == "-")
    if not channel_name:
        channel_name = "unnamed-agent"

    try:
        channel = await guild.create_text_channel(channel_name)
        logger.info(f"Auto-created channel #{channel_name} (ID: {channel.id}) for agent {machine_name}")

        # Update config mapping
        self.bot_config.channel_mapping[str(channel.id)] = machine_name
        await self.bot_config.save_config()

        # Send connect notification in new channel
        await channel.send(f"\U0001f7e2 **{machine_name}** connected ({platform})\n_Channel auto-created for this machine._")

    except discord.Forbidden:
        logger.error(f"Missing manage_channels permission in guild {guild_id}, cannot auto-create channel for {machine_name}")
    except Exception as e:
        logger.error(f"Failed to auto-create channel for {machine_name}: {e}")
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Discord user -> Slash command | User identity from Discord, checked against allowlist |
| Bot -> Guild channel | Bot creates channels; needs manage_channels permission |
| Config.json -> Bot | Reload reads local file; could be tampered if machine is compromised |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-05 | Elevation of Privilege | cog interaction_check | mitigate | Every cog has interaction_check verifying user.id in allowed_user_ids; guild_id checked at tree level |
| T-02-06 | Spoofing | interaction.user.id | accept | Discord guarantees user identity via OAuth; we trust Discord's user.id |
| T-02-07 | Denial of Service | /reload command | mitigate | /reload only available to users in allowed_user_ids (already admin-equivalent); re-sync is bounded by guild count |
| T-02-08 | Tampering | auto-channel config write | accept | Config.json is local; auto-channel creation only adds entries, never removes; asyncio.Lock prevents corruption |
| T-02-09 | Information Disclosure | /status agent info | accept | Only visible to allowed users in allowed guilds; agent name and platform are not sensitive |
</threat_model>
