```python
"""Tests for channel-to-agent routing logic (ROUT-01, ROUT-03)."""
import pytest

# Test get_agent_for_channel logic (extracted from TerminalBot)
class TestChannelRouting:
    def test_mapped_channel_returns_machine_name(self):
        """ROUT-01: channel maps to one agent."""
        mapping = {"111222333": "my-desktop", "444555666": "my-laptop"}
        channel_id = 111222333
        result = mapping.get(str(channel_id))
        assert result == "my-desktop"

    def test_unmapped_channel_returns_none(self):
        """ROUT-03: unmapped channel returns None."""
        mapping = {"111222333": "my-desktop"}
        channel_id = 999888777
        result = mapping.get(str(channel_id))
        assert result is None

    def test_channel_id_int_vs_string(self):
        """Pitfall 3: channel IDs are int from Discord, string keys in config."""
        mapping = {"111222333": "my-desktop"}
        # Discord gives int, config has string keys
        assert mapping.get(str(111222333)) == "my-desktop"
        assert mapping.get(111222333) is None  # int key fails

    def test_reverse_lookup_finds_channel(self):
        """Find channel for a machine_name (reverse mapping)."""
        mapping = {"111222333": "my-desktop", "444555666": "my-laptop"}
        # Simulate _find_channel_for_agent logic
        machine_name = "my-laptop"
        result = None
        for ch_id, name in mapping.items():
            if name == machine_name:
                result = ch_id
                break
        assert result == "444555666"

    def test_reverse_lookup_no_match(self):
        """Reverse lookup returns None for unknown machine."""
        mapping = {"111222333": "my-desktop"}
        machine_name = "unknown-machine"
        result = None
        for ch_id, name in mapping.items():
            if name == machine_name:
                result = ch_id
                break
        assert result is None
```

```python
        """Empty agent dict produces gray embed with 'No agents connected'."""
        import discord
        agents = {}
        embed = discord.Embed(
            title="Agent Status",
            description="No agents connected",
            color=0x95A5A6,
        )
        assert embed.title == "Agent Status"
        assert embed.description == "No agents connected"
        assert embed.color.value == 0x95A5A6

        """Agent dict produces green embed with fields per agent."""
        import discord
        embed = discord.Embed(title="Agent Status", color=0x2ECC71)
        # Simulate adding agents
        agents = {
            "my-desktop": MagicMock(platform="win32", last_pong=time.time()),
            "my-laptop": MagicMock(platform="darwin", last_pong=time.time()),
        }
        for name, info in agents.items():
            embed.add_field(
                name=f"\U0001f7e2 {name}",
                value=f"**Platform:** {info.platform}\n**Status:** Online",
                inline=False,
            )
        assert embed.color.value == 0x2ECC71
        assert len(embed.fields) == 2
        assert "my-desktop" in embed.fields[0].name
        assert "win32" in embed.fields[0].value
        assert "my-laptop" in embed.fields[1].name

    def test_status_footer_shows_count(self):
        """Footer shows agent count."""
        import discord
        embed = discord.Embed(title="Agent Status", color=0x2ECC71)
```

```python
class TestAgentManagerCallbacks:

        config = MagicMock()

        """Callbacks can be assigned as async functions."""
        config = MagicMock()
        config.shared_secret = "test"
        config.heartbeat_interval = 15
        config.heartbeat_timeout = 30
        config.registration_timeout = 10
        manager = AgentManager(config)

        connect_cb = AsyncMock()
        disconnect_cb = AsyncMock()
        manager.on_agent_connect = connect_cb
        manager.on_agent_disconnect = disconnect_cb

        assert manager.on_agent_connect is connect_cb
        assert manager.on_agent_disconnect is disconnect_cb


class TestChannelNameSanitization:
    """Test auto-channel name generation."""

    @pytest.mark.parametrize("input_name,expected", [
        ("my-desktop", "my-desktop"),
        ("My Desktop", "my-desktop"),
        ("my_laptop", "my-laptop"),
        ("WORK PC", "work-pc"),
        ("test@machine!", "testmachine"),
    ])
    def test_channel_name_sanitization(self, input_name, expected):
        """Channel names are lowercase, alphanumeric + hyphens only."""
        sanitized = input_name.lower().replace(" ", "-").replace("_", "-")
        sanitized = "".join(c for c in sanitized if c.isalnum() or c == "-")
        assert sanitized == expected
```
